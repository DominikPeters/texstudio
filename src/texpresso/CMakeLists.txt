# Can also be built separately for fast protocol/session integration tests.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TexpressoClient LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_AUTOMOC ON)
    include(GNUInstallDirs)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets Concurrent Test)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets Concurrent Test)
    option(TEXSTUDIO_TEXPRESSO_TESTS "Build live-preview tests" ON)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../pdfviewer/ViewerComponents.cmake")
add_library(texpresso-client STATIC
    texpressoprotocol.cpp texpressoprotocol.h
    texpressoprocess.cpp texpressoprocess.h
    texpressopreview.cpp texpressopreview.h
    texpressosession.cpp texpressosession.h texpressosourcefollow.h
    texpressomanager.cpp texpressomanager.h
    texpressoprewarm.cpp texpressoprewarm.h
    texpressowidget.cpp texpressowidget.h)
target_include_directories(texpresso-client PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(texpresso-client PUBLIC document-viewer Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
if(NOT APPLE)
    if(NOT CMAKE_INSTALL_LIBEXECDIR)
        set(CMAKE_INSTALL_LIBEXECDIR libexec)
    endif()
    file(RELATIVE_PATH _txp_install_relative "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBEXECDIR}/texstudio/texpresso")
    target_compile_definitions(texpresso-client PRIVATE TEXPRESSO_INSTALL_RELATIVE="${_txp_install_relative}")
endif()
option(TEXSTUDIO_TEXPRESSO_TESTS "Build live-preview tests" OFF)
if(TEXSTUDIO_TEXPRESSO_TESTS)
    enable_testing()
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(texpresso-client-tests tests/client_test.cpp)
    target_link_libraries(texpresso-client-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-client COMMAND texpresso-client-tests)
    add_executable(texpresso-rendering-tests tests/rendering_test.cpp)
    target_link_libraries(texpresso-rendering-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-rendering COMMAND texpresso-rendering-tests)
    set_tests_properties(texpresso-rendering PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(texpresso-compiler-tests tests/compiler_test.cpp)
    target_link_libraries(texpresso-compiler-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-compiler COMMAND texpresso-compiler-tests)
    set_tests_properties(texpresso-compiler PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(texpresso-diagnostics-tests tests/diagnostics_test.cpp)
    target_link_libraries(texpresso-diagnostics-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-diagnostics COMMAND texpresso-diagnostics-tests)

    add_test(NAME texpresso-rerun COMMAND texpresso-rerun-tests)
    set_tests_properties(texpresso-rerun PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(texpresso-scrolling-tests tests/scrolling_test.cpp)
    target_link_libraries(texpresso-scrolling-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)

    target_link_libraries(texpresso-prewarm-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-prewarm COMMAND texpresso-prewarm-tests)

    target_link_libraries(texpresso-links-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME texpresso-links COMMAND texpresso-links-tests)
    set_tests_properties(texpresso-links PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

        string(REPLACE "-" "_" _source "${_test}")
        if(_test STREQUAL "sustained")
            set(_source sustained_editing)
        endif()
        add_executable(texpresso-${_test}-tests tests/${_source}_test.cpp)
        target_link_libraries(texpresso-${_test}-tests PRIVATE texpresso-client Qt${QT_VERSION_MAJOR}::Test)

        set_tests_properties(texpresso-${_test} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 600)
    endforeach()
    target_compile_definitions(texpresso-sustained-tests PRIVATE TEXPRESSO_SUSTAINED_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/sustained")
    set_tests_properties(texpresso-client PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
